// Mini driver for UI screenshot verification: shows the real chat form on an STA
// thread with a message loop, exposes bounds for CopyFromScreen, then closes.
public static class ChatUiShot
{
    public static int FormX, FormY, FormW, FormH;
    public static bool FormShown;
    public static string Err;

    public static void Show()
    {
        var t = new System.Threading.Thread(delegate() {
            try {
                typeof(ChatPlugin).GetMethod("Run", System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Static).Invoke(null, null);
                var f = (System.Windows.Forms.Form)typeof(ChatPlugin).GetField("chatForm", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static).GetValue(null);
                f.Shown += delegate {
                    var b = f.Bounds;
                    FormX = b.X; FormY = b.Y; FormW = b.Width; FormH = b.Height; FormShown = true;
                };
                System.Windows.Forms.Application.Run();
            } catch (System.Exception ex) { Err = ex.Message; }
        });
        t.SetApartmentState(System.Threading.ApartmentState.STA);
        t.IsBackground = true;
        t.Start();
    }

    public static void Close()
    {
        try {
            var f = (System.Windows.Forms.Form)typeof(ChatPlugin).GetField("chatForm", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static).GetValue(null);
            if (f != null) f.Invoke(new System.Action(delegate { f.Close(); }));
        } catch {}
    }
}
