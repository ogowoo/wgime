// C# driver appended after the plugin source: drives ChatPlugin headlessly and records
// all Notify* callbacks in-process (no PowerShell delegates on foreign threads).
public static class ChatPluginDriver
{
    public static System.Collections.Generic.List<string> Log = new System.Collections.Generic.List<string>();
    static System.Reflection.BindingFlags F = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Static;
    static System.Type T = typeof(ChatPlugin);
    static void SetF(string n, object v) { T.GetField(n, F).SetValue(null, v); }
    static object GetF(string n) { return T.GetField(n, F).GetValue(null); }
    static void CallM(string n, params object[] args) { T.GetMethod(n, F).Invoke(null, args); }
    static void Add(string s) { lock (Log) Log.Add(s); }

    class InlineSync : System.Threading.SynchronizationContext
    {
        public override void Post(System.Threading.SendOrPostCallback d, object state) { try { d(state); } catch (System.Exception ex) { Add("POST-ERR|" + ex.Message); } }
        public override void Send(System.Threading.SendOrPostCallback d, object state) { try { d(state); } catch {} }
    }

    public static string[] Run(string brokerUrl, string room, string nick, string key, int waitMs, string sendText)
    {
        SetF("nick", nick); SetF("room", room); SetF("customKey", key);
        SetF("brokerSel", brokerUrl); SetF("lastBroker", "");
        SetF("docId", "harn-" + System.Guid.NewGuid().ToString("N").Substring(0, 8));
        SetF("uiContext", new InlineSync());
        SetF("manualLeave", false); SetF("relayMode", false);
        // dummy UI controls for failure paths
        SetF("edNick", new System.Windows.Forms.TextBox());
        SetF("edRoom", new System.Windows.Forms.TextBox());
        SetF("edKey", new System.Windows.Forms.TextBox());
        SetF("edBroker", new System.Windows.Forms.ComboBox());
        var flat = T.GetNestedType("FlatBtn", System.Reflection.BindingFlags.NonPublic);
        SetF("joinBtn", System.Activator.CreateInstance(flat));
        SetF("onMessage", new System.Action<string, string, string>(delegate(string n, string t2, string ts) { Add("MSG|" + n + "|" + ts + "|" + t2); }));
        SetF("onEvent", new System.Action<string>(delegate(string s) { Add("EVT|" + s); }));
        SetF("onStatus", new System.Action<string, bool>(delegate(string s, bool ok) { Add("STATUS|" + ok + "|" + s); }));
        SetF("onTyping", new System.Action<string>(delegate(string s) { Add("TYPING|" + s); }));
        SetF("onUsers", new System.Action(delegate { Add("USERS"); }));
        SetF("onRooms", new System.Action(delegate { }));

        var th = new System.Threading.Thread(delegate() { CallM("ConnectWorker"); });
        th.IsBackground = true; th.Start();

        var deadline = System.DateTime.Now.AddSeconds(45);
        while (!(bool)GetF("running") && System.DateTime.Now < deadline) System.Threading.Thread.Sleep(300);
        Add("HARNESS: running=" + GetF("running"));
        if ((bool)GetF("running")) {
            System.Threading.Thread.Sleep(2000);
            CallM("SendChat", sendText);
            Add("HARNESS: sent " + sendText);
        }
        System.Threading.Thread.Sleep(waitMs);
        lock (Log) return Log.ToArray();
    }
}
