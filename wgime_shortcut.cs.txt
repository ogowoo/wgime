    // ---------- launcher shortcut (per-machine; IShellLink primary, WScript.Shell fallback) ----------
    // Creates "<batname>.lnk" next to the bat targeting powershell.exe DIRECTLY
    // (no bat/cmd involved): loads the DLL and runs - zero console flash, and
    // -Command is not ExecutionPolicy-gated (works under Restricted policy).
    // The icon comes from the DLL's embedded resource. Never committed.
    // An existing shortcut is REUSED only if its arguments still point at THIS
    // machine's DLL; otherwise (copied folder, old .bat target, unreadable)
    // it is rebuilt. All failures are logged to %TEMP%\__ERRLOG__.
    [System.Runtime.InteropServices.ComImport, System.Runtime.InteropServices.Guid("00021401-0000-0000-C000-000000000046")]
    class ShellLinkCom { }
    [System.Runtime.InteropServices.ComImport,
     System.Runtime.InteropServices.InterfaceType(System.Runtime.InteropServices.ComInterfaceType.InterfaceIsIUnknown),
     System.Runtime.InteropServices.Guid("000214F9-0000-0000-C000-000000000046")]
    interface IShellLinkW
    {
        void GetPath(System.Text.StringBuilder pszFile, int cch, IntPtr pfd, int fFlags);
        void GetIDList(out IntPtr ppidl);
        void SetIDList(IntPtr pidl);
        void GetDescription(System.Text.StringBuilder pszName, int cch);
        void SetDescription(string pszName);
        void GetWorkingDirectory(System.Text.StringBuilder pszDir, int cch);
        void SetWorkingDirectory(string pszDir);
        void GetArguments(System.Text.StringBuilder pszArgs, int cch);
        void SetArguments(string pszArgs);
        void GetHotkey(out short pwHotkey);
        void SetHotkey(short wHotkey);
        void GetShowCmd(out int piShowCmd);
        void SetShowCmd(int iShowCmd);
        void GetIconLocation(System.Text.StringBuilder pszIconPath, int cch, out int piIcon);
        void SetIconLocation(string pszIconPath, int iIcon);
        void SetRelativePath(string pszPathRel, int dwReserved);
        void Resolve(IntPtr hwnd, int fFlags);
        void SetPath(string pszFile);
    }
    [System.Runtime.InteropServices.ComImport,
     System.Runtime.InteropServices.InterfaceType(System.Runtime.InteropServices.ComInterfaceType.InterfaceIsIUnknown),
     System.Runtime.InteropServices.Guid("0000010B-0000-0000-C000-000000000046")]
    interface IPersistFile
    {
        void GetClassID(out Guid pClassID);
        void IsDirty();
        void Load(string pszFileName, int dwMode);
        void Save(string pszFileName, bool fRemember);
        void SaveCompleted(string pszFileName);
        void GetCurFile(out string pszFileName);
    }
    static string ScSq(string s) { return (s == null ? "" : s).Replace("'", "''").Replace("$", ((char)96).ToString() + "$"); }
    static void ScLog(string m)
    {
        try { File.AppendAllText(Path.Combine(Path.GetTempPath(), "__ERRLOG__"), "[" + DateTime.Now.ToString("s") + "] shortcut: " + m + "\r\n", Encoding.UTF8); } catch {}
    }
    static bool ScWrite(string lnk, string target, string args, string dir, string desc, string iconPath)
    {
        try {
            var sl = (IShellLinkW)(new ShellLinkCom());
            sl.SetPath(target);
            sl.SetArguments(args);
            sl.SetWorkingDirectory(dir);
            sl.SetDescription(desc);
            sl.SetIconLocation(iconPath, 0);
            sl.SetShowCmd(7);
            ((IPersistFile)sl).Save(lnk, true);
            return true;
        } catch (Exception e1) {
            try {
                var t = Type.GetTypeFromProgID("WScript.Shell");
                if (t == null) throw new Exception("no WSH ProgID");
                var sh = Activator.CreateInstance(t);
                var s = t.InvokeMember("CreateShortcut", System.Reflection.BindingFlags.InvokeMethod, null, sh, new object[] { lnk });
                var st = s.GetType();
                st.InvokeMember("TargetPath", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { target });
                st.InvokeMember("Arguments", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { args });
                st.InvokeMember("WorkingDirectory", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { dir });
                st.InvokeMember("Description", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { desc });
                st.InvokeMember("IconLocation", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { iconPath + ",0" });
                st.InvokeMember("WindowStyle", System.Reflection.BindingFlags.SetProperty, null, s, new object[] { 7 });
                st.InvokeMember("Save", System.Reflection.BindingFlags.InvokeMethod, null, s, null);
                return true;
            } catch (Exception e2) { ScLog("write failed (IShellLink: " + e1.Message + " | WSH: " + e2.Message + ")"); }
        }
        return false;
    }
    static bool ScCheck(string lnk, string dll)
    {
        // true = existing shortcut already targets THIS dll (reuse it)
        try {
            var sl = (IShellLinkW)(new ShellLinkCom());
            ((IPersistFile)sl).Load(lnk, 0);
            var sbA = new System.Text.StringBuilder(8192);
            sl.GetArguments(sbA, sbA.Capacity);
            if (sbA.ToString().IndexOf(dll, StringComparison.OrdinalIgnoreCase) < 0) return false;   // wrong machine / old path -> rebuild
            var sbT = new System.Text.StringBuilder(1024);
            sl.GetPath(sbT, sbT.Capacity, IntPtr.Zero, 0);
            return sbT.ToString().EndsWith("powershell.exe", StringComparison.OrdinalIgnoreCase);
        } catch { return false; }   // unreadable / old format -> rebuild
    }
    static void EnsureShortcut(string dir, string batPath)
    {
        try {
            string dll = System.Reflection.Assembly.GetExecutingAssembly().Location;
            string lnk = Path.Combine(dir, Path.GetFileNameWithoutExtension(batPath) + ".lnk");
            if (File.Exists(lnk) && ScCheck(lnk, dll)) return;   // already points at this machine's DLL
            string args = "-NoProfile -NoLogo -STA -WindowStyle Hidden -Command \"try { Add-Type -AssemblyName System.Windows.Forms; Add-Type -Path '" + ScSq(dll) + "'; __ENTRY__::Run('" + ScSq(dir) + "', '" + ScSq(batPath) + "') } catch { [IO.File]::WriteAllText((Join-Path `$env:TEMP '__ERRLOG__'), (`$_ | Out-String)); [System.Windows.Forms.MessageBox]::Show((`$_ | Out-String),'__DESC__ Error') | Out-Null }\"";
            if (ScWrite(lnk, "powershell.exe", args, dir, "__DESC__", dll)) return;
            // bat dir not writable (Program Files etc.): fall back to the user desktop
            string fb = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory), Path.GetFileName(lnk));
            if (!fb.Equals(lnk, StringComparison.OrdinalIgnoreCase) && ScWrite(fb, "powershell.exe", args, dir, "__DESC__", dll))
                ScLog("dir not writable - shortcut created on Desktop: " + fb);
            else
                ScLog("could not create shortcut at " + lnk + " nor on the Desktop");
        } catch (Exception ex) { ScLog("ensure failed: " + ex.Message); }
    }
