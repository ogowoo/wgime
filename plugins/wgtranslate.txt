    static void StartTranslate() {
        if(_busy || _input==null || _input.Text.Trim().Length==0) return;
        string text=_input.Text; bool chinese=Regex.IsMatch(text,"[\u4e00-\u9fff]");

        if(src=="auto") src=chinese?"zh-CN":"en";
        if(dst=="auto-bi") { dst=chinese?"en":"zh-CN"; _target.SelectedItem="自动中英互译"; }
        if(src==dst) { SetStatus("源语言和目标语言不能相同"); return; }
        string provider=(string)_provider.SelectedItem;
        _busy=true; _go.Enabled=false; _clear.Enabled=false; _output.Text=""; SetStatus("正在翻译...");
        ThreadPool.QueueUserWorkItem(delegate {
            try {
                List<string> chunks=SplitUtf8(text,450); StringBuilder all=new StringBuilder(); string used=provider;
                for(int i=0;i<chunks.Count;i++) { int n=i+1; _form.BeginInvoke((MethodInvoker)delegate { SetStatus("正在翻译 "+n+"/"+chunks.Count+" · "+used); }); string actual; all.Append(Translate(chunks[i],src,dst,provider,out actual)); used=actual; }
                string result=all.ToString(), finalUsed=used; _form.BeginInvoke((MethodInvoker)delegate { _output.Text=result; SetStatus("完成 · "+finalUsed); _busy=false; _go.Enabled=true; _clear.Enabled=true; });
            } catch(Exception ex) { _form.BeginInvoke((MethodInvoker)delegate { _output.Text="翻译失败：\r\n"+ex.Message; SetStatus("翻译失败"); _busy=false; _go.Enabled=true; _clear.Enabled=true; }); }

    }
    static List<string> SplitUtf8(string s,int max) { List<string> a=new List<string>(); StringBuilder b=new StringBuilder(); foreach(char c in s) { string x=b.ToString()+c; if(Encoding.UTF8.GetByteCount(x)>max && b.Length>0) { a.Add(b.ToString()); b.Length=0; } b.Append(c); } if(b.Length>0)a.Add(b.ToString()); return a; }
    static string Translate(string q,string src,string dst,string provider,out string used) {
        if(provider!="自动容错") { used=provider; return Call(provider,q,src,dst); }
        string[] ps={"Google 翻译（网页）","MyMemory","Lingva 公共实例","Google Cloud（API Key）","LibreTranslate 本机"}; StringBuilder err=new StringBuilder();
        foreach(string p in ps) try { string r=Call(p,q,src,dst); used=p; return r; } catch(Exception ex) { err.Append(p+": "+ex.Message+" | "); }
        throw new Exception("所有接口均失败："+err.ToString());
    }
    static string Call(string p,string q,string src,string dst) { if(p=="Google 翻译（网页）")return GoogleWeb(q,src,dst); if(p=="Google Cloud（API Key）")return GoogleCloud(q,src,dst); if(p=="MyMemory")return MyMemory(q,src,dst); if(p=="Lingva 公共实例")return Lingva(q,src,dst); return Libre(q,src,dst); }
    static string Http(string url,string method,string body) {

    }
    static string JEsc(string s) { return s.Replace("\\","\\\\").Replace("\"","\\\"").Replace("\r","\\r").Replace("\n","\\n").Replace("\t","\\t"); }
    static string JVal(string json,string key) {
        Match m=Regex.Match(json,"\\\""+Regex.Escape(key)+"\\\"\\s*:\\s*\\\"((?:\\\\.|[^\\\"])*)\\\"",RegexOptions.Singleline);
        if(!m.Success) throw new Exception("响应缺少 "+key);
        string v=m.Groups[1].Value.Replace("\\/","/").Replace("\\\"","\"").Replace("\\\\","\\");
        return Regex.Replace(v,"\\\\u([0-9a-fA-F]{4})",delegate(Match x){return ((char)Convert.ToInt32(x.Groups[1].Value,16)).ToString();});
    }

    public class KeyboardHook {
        const int WH_KEYBOARD_LL=13,WM_KEYDOWN=0x100,WM_SYSKEYDOWN=0x104,VK_C=0x43,VK_CONTROL=0x11;
        delegate IntPtr Proc(int n,IntPtr w,IntPtr l); Proc _proc; IntPtr _h; Action _fire; long _last;
        [StructLayout(LayoutKind.Sequential)] struct K { public uint vkCode,scanCode,flags,time; public IntPtr extra; }
        [DllImport("user32.dll")] static extern IntPtr SetWindowsHookEx(int id,Proc p,IntPtr m,uint t); [DllImport("user32.dll")] static extern bool UnhookWindowsHookEx(IntPtr h); [DllImport("user32.dll")] static extern IntPtr CallNextHookEx(IntPtr h,int n,IntPtr w,IntPtr l); [DllImport("user32.dll")] static extern short GetAsyncKeyState(int k);
        public KeyboardHook(Action a){_fire=a;_proc=Cb;} public void Start(){_h=SetWindowsHookEx(WH_KEYBOARD_LL,_proc,IntPtr.Zero,0);} public void Stop(){if(_h!=IntPtr.Zero)UnhookWindowsHookEx(_h);_h=IntPtr.Zero;}
        IntPtr Cb(int n,IntPtr w,IntPtr l){if(n>=0&&(w==(IntPtr)WM_KEYDOWN||w==(IntPtr)WM_SYSKEYDOWN)){K k=(K)Marshal.PtrToStructure(l,typeof(K));if(k.vkCode==VK_C&&(GetAsyncKeyState(VK_CONTROL)&0x8000)!=0){long now=Environment.TickCount;if(_last!=0&&now-_last>0&&now-_last<=650){_last=0;if(_fire!=null)_fire();}else _last=now;}}return CallNextHookEx(_h,n,w,l);}
    }
