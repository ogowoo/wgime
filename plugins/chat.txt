    // ---------- crypto (compatible with chat.bat: SHA256 key, AES-256-CBC, HMAC) ----------
    static byte[] DeriveKey()
    {
        string raw = room + ":" + (customKey.Length > 0 ? customKey : room);
        using (var sha = SHA256.Create()) return sha.ComputeHash(Encoding.UTF8.GetBytes(raw));
    }
    static string Encrypt(string plain)
    {
        try {
            byte[] key = DeriveKey();
            byte[] iv = new byte[16];
            using (var rng = new RNGCryptoServiceProvider()) rng.GetBytes(iv);
            using (var aes = Aes.Create()) {
                aes.Key = key; aes.IV = iv; aes.Mode = CipherMode.CBC; aes.Padding = PaddingMode.PKCS7;
                using (var enc = aes.CreateEncryptor()) {
                    byte[] pt = Encoding.UTF8.GetBytes(plain);
                    byte[] ct = enc.TransformFinalBlock(pt, 0, pt.Length);
                    string ivHex = ToHex(iv), ctHex = ToHex(ct);
                    string hmacInput = ivHex + ":" + ctHex;
                    string hmac;
                    using (var h = new HMACSHA256(key)) hmac = ToHex(h.ComputeHash(Encoding.UTF8.GetBytes(hmacInput)));
                    return ivHex + ":" + ctHex + ":" + hmac;
                }
            }
        } catch { return plain; }
    }
    static string Decrypt(string data)
    {
        try {
            byte[] key = DeriveKey();
            var p = data.Split(':');
            if (p.Length < 3) return null;
            string ivHex = p[0], ctHex = p[1], hmacStr = p[2];
            string hmacInput = ivHex + ":" + ctHex;
            string expect;
            using (var h = new HMACSHA256(key)) expect = ToHex(h.ComputeHash(Encoding.UTF8.GetBytes(hmacInput)));
            if (expect != hmacStr) return null;
            using (var aes = Aes.Create()) {
                aes.Key = key; aes.IV = FromHex(ivHex); aes.Mode = CipherMode.CBC; aes.Padding = PaddingMode.PKCS7;
                using (var dec = aes.CreateDecryptor()) {
                    byte[] ct = FromHex(ctHex);
                    byte[] pt = dec.TransformFinalBlock(ct, 0, ct.Length);
                    return Encoding.UTF8.GetString(pt);
                }
            }
        } catch { return null; }
    }
    static string ToHex(byte[] b) { var sb = new StringBuilder(); foreach (byte x in b) sb.Append(x.ToString("x2")); return sb.ToString(); }
    static byte[] FromHex(string hex)
    {
        var b = new byte[hex.Length / 2];
        for (int i = 0; i < b.Length; i++) b[i] = Convert.ToByte(hex.Substring(i * 2, 2), 16);
        return b;
    }

    // ---------- MQTT frame builders (CONNECT/SUBSCRIBE/PUBLISH/PINGREQ/DISCONNECT) ----------
    static void MqttConnect()
    {
        var payload = new List<byte>();
        AddMqttString(payload, "MQTT");                        // protocol name
        payload.Add(4);                                        // protocol level 4 (MQTT 3.1.1)
        payload.Add(2);                                        // clean session
        AddMqttU16(payload, 30);                               // keepalive 30s
        AddMqttString(payload, docId);                         // client id
        WsSend(BuildMqtt(0x10, payload.ToArray()), WebSocketMessageType.Binary);
    }
    static void MqttSubscribe(string topic)
    {
        mqttPktId = (mqttPktId % 60000) + 1;
        var payload = new List<byte>();
        AddMqttU16(payload, mqttPktId);                        // packet id
        AddMqttString(payload, topic);
        payload.Add(0);                                        // qos 0
        WsSend(BuildMqtt(0x82, payload.ToArray()), WebSocketMessageType.Binary);
    }
    static void MqttPublish(string topic, string json)
    {
        var payload = new List<byte>();
        AddMqttString(payload, topic);
        payload.AddRange(Encoding.UTF8.GetBytes(json));
        WsSend(BuildMqtt(0x30, payload.ToArray()), WebSocketMessageType.Binary);
    }
    static void MqttPing() { WsSend(new byte[] { 0xC0, 0x00 }, WebSocketMessageType.Binary); }
    static void MqttDisconnect() { try { if (!relayMode) WsSend(new byte[] { 0xE0, 0x00 }, WebSocketMessageType.Binary); } catch {} }

    static byte[] BuildMqtt(byte type, byte[] payload)
    {
        var pkt = new List<byte>();
        pkt.Add(type);
        int rem = payload.Length;
        do {
            byte b = (byte)(rem % 128);
            rem /= 128;
            if (rem > 0) b |= 0x80;
            pkt.Add(b);
        } while (rem > 0);
        pkt.AddRange(payload);
        return pkt.ToArray();
    }
    static void AddMqttString(List<byte> buf, string s)
    {
        byte[] b = Encoding.UTF8.GetBytes(s);
        buf.Add((byte)(b.Length >> 8)); buf.Add((byte)(b.Length & 0xFF));
        buf.AddRange(b);
    }
    static void AddMqttU16(List<byte> buf, int v) { buf.Add((byte)(v >> 8)); buf.Add((byte)(v & 0xFF)); }

    // ---------- connect / disconnect (全部在后台线程) ----------
    static void ConnectWorker()
    {
        bool auto = IsAuto(brokerSel);
        string[] tryUrls;
        if (auto) {
            var list = new List<string>();
            if (lastBroker.Length > 0) list.Add(lastBroker);
            foreach (string u in BrokerUrls) if (!list.Contains(u)) list.Add(u);
            tryUrls = list.ToArray();
        } else {
            tryUrls = new string[] { brokerSel };
        }
        foreach (string url in tryUrls) {
            if (manualLeave) { connecting = false; return; }
            brokerUrl = url;
            relayMode = IsRelay(url);
            NotifyStatus("连接中 " + url + " …", false);
            try {
                WsClose();
                cts = new CancellationTokenSource();
                ws = new ClientWebSocket();
                ws.Options.KeepAliveInterval = TimeSpan.FromSeconds(25);
                if (!relayMode) ws.Options.AddSubProtocol("mqtt");    // EMQX/Mosquitto 要求 mqtt 子协议
                string full = relayMode ? url.TrimEnd('/') + "/room/" + Uri.EscapeDataString(room) : MqttUrl(url);
                // 10s 连接超时: WhenAny 兜底 (.NET 4.x 的 token 取消不一定及时)
                var connTask = ws.ConnectAsync(new Uri(full), cts.Token);
                if (System.Threading.Tasks.Task.WhenAny(connTask, System.Threading.Tasks.Task.Delay(10000)).Result != connTask) {
                    try { ws.Abort(); } catch {}
                    throw new Exception("connect timeout");
                }
                connTask.Wait();
                if (relayMode) {
                    JoinComplete();
                    ReceiveLoop();                                     // 阻塞到断线, 内部走 OnDisconnected
                } else {
                    connackEvent = new ManualResetEvent(false);
                    MqttConnect();
                    var rt = new Thread(ReceiveLoop); rt.IsBackground = true; rt.Start();
                    if (!connackEvent.WaitOne(8000)) throw new Exception("CONNACK timeout");
                    // CONNACK 里已 MqttSubscribe + JoinComplete
                }
                connecting = false;
                return;
            } catch (Exception ex) {
                WsClose();
                if (!auto) {
                    string m = ex is AggregateException && ex.InnerException != null ? ex.InnerException.Message : ex.Message;
                    NotifyStatus("连接失败: " + m, false);
                    UiJoinFailed();
                    return;
                }
                NotifyStatus(url + " 失败, 尝试下一个…", false);
            }
        }
        connecting = false;
        if (!manualLeave) { NotifyStatus("所有服务器均连接失败", false); UiJoinFailed(); }
    }

    static void OnConnack()
    {
        try { if (connackEvent != null) connackEvent.Set(); } catch {}
        MqttSubscribe(Topic());
        MqttSubscribe(REGISTRY_TOPIC);
        JoinComplete();
    }

    static void JoinComplete()
    {
        running = true;
        connecting = false;
        reconnectTries = 0;
        if (IsAuto(brokerSel) && brokerUrl != lastBroker) { lastBroker = brokerUrl; SaveCfg(); }
        SendJoin();
        NotifyStatus("已连接 " + (relayMode ? "[中继] " : "[MQTT] ") + brokerUrl, true);
        NotifyEvent("已加入房间 [" + room + "] (" + (relayMode ? "Cloudflare 中继" : "MQTT") + ")");
        StartTimers();
    }

    static void ReceiveLoop()
    {
        var ms = new MemoryStream();
        var buf = new byte[65536];
        while (true) {
            WebSocketReceiveResult res;
            try {
                var w = ws; var c = cts;
                if (w == null || c == null || w.State != WebSocketState.Open) break;
                res = w.ReceiveAsync(new ArraySegment<byte>(buf), c.Token).Result;
            } catch { break; }
            if (res.MessageType == WebSocketMessageType.Close) break;
            try { ms.Write(buf, 0, res.Count); } catch { break; }
            if (!res.EndOfMessage) continue;                          // 拼 WS 分片
            byte[] frame = ms.ToArray(); ms.SetLength(0);
            try {
                if (relayMode) {
                    if (res.MessageType == WebSocketMessageType.Text) {
                        string js = Encoding.UTF8.GetString(frame);
                        Dbg("RX[relay]", js);
                        OnJsonMessage(js);
                    }
                } else {
                    if (res.MessageType == WebSocketMessageType.Binary)
                        HandleMqttPackets(frame);
                }
            } catch {}
        }
        OnDisconnected();
    }

    static void HandleMqttPackets(byte[] frame)
    {
        int pos = 0;
        while (pos + 2 <= frame.Length) {
            byte head = frame[pos];
            int type = head & 0xF0;
            int i = pos + 1, rem = 0, shift = 0, guard = 0;
            bool trunc = false;
            while (true) {
                if (i >= frame.Length || ++guard > 4) { trunc = true; break; }
                byte b = frame[i++];
                rem |= (b & 0x7F) << shift; shift += 7;
                if ((b & 0x80) == 0) break;
            }
            if (trunc || i + rem > frame.Length) return;              // 不完整帧, 丢弃
            int bodyStart = i;
            if (type == 0x20) {                                       // CONNACK
                if (rem >= 2 && frame[i + 1] == 0) OnConnack();
            } else if (type == 0x30 || type == 0x32 || type == 0x34) { // PUBLISH qos 0/1/2
                int qos = (head >> 1) & 0x3;
                if (rem >= 2) {
                    int tl = (frame[i] << 8) | frame[i + 1];
                    if (2 + tl <= rem) {
                        string topic = Encoding.UTF8.GetString(frame, i + 2, tl);
                        int p = i + 2 + tl;
                        if (qos > 0) p += 2;                          // 跳过 packet id
                        int plen = i + rem - p;
                        if (plen >= 0) DispatchJson(topic, Encoding.UTF8.GetString(frame, p, plen));
                    }
                }
            }
            // SUBACK(0x90) / PINGRESP(0xD0) / 其他: 忽略
            pos = bodyStart + rem;
        }
    }

    static void DispatchJson(string topic, string json)
    {
        Dbg("RX[" + topic + "]", json);
        if (topic == REGISTRY_TOPIC) { OnRegistryMessage(json); return; }
        OnJsonMessage(json);
    }

    static void OnDisconnected()
    {
        bool wasRunning = running;
        running = false;
        connecting = false;
        StopTimers();
        WsClose();
        if (manualLeave) return;
        if (!wasRunning) { NotifyStatus("已断开", false); return; }
        NotifyEvent("连接已断开");
        if (reconnectTries < 3) {
            reconnectTries++;
            NotifyStatus("已断开, 6 秒后重连 (" + reconnectTries + "/3)…", false);
            var t = new Thread(delegate() {
                Thread.Sleep(6000);
                if (!manualLeave) { connecting = true; ConnectWorker(); }
            });
            t.IsBackground = true; t.Start();
        } else {
            NotifyStatus("重连失败, 已断开", false);
            UiJoinFailed();
        }
    }

    // ---------- protocol messages (itools-chat compatible) ----------
    static void OnJsonMessage(string json)
    {
        string type = JsonGet(json, "type");
        if (type == null) return;
        string rid = JsonGet(json, "id");
        string rnick = JsonGet(json, "nick");
        string rts = JsonGet(json, "ts");

        if (type == "join") {
            if (rid != docId && rnick != null) {
                users[rid] = rnick; NotifyEvent(rnick + " 加入了"); NotifyUsers();
                SendJson("{\"type\":\"online\",\"nick\":\"" + JsonEsc(nick) + "\",\"ts\":" + DateTimeOffset.Now.ToUnixTimeMilliseconds() + ",\"id\":\"" + JsonEsc(docId) + "\"}");
            }
        } else if (type == "leave") {
            if (users.Remove(rid)) { NotifyEvent((rnick ?? "?") + " 离开了"); NotifyUsers(); }
        } else if (type == "chat") {
            if (rid == docId) return;
            string key = "m" + rnick + ":" + rts;
            if (recentMsgs.Contains(key)) return;
            recentMsgs.Add(key); if (recentMsgs.Count > 300) recentMsgs.Clear();
            string pt = Decrypt(JsonGet(json, "text"));
            if (pt == null) { NotifyMessage(rnick ?? "?", "[encrypted]", rts, null); return; }   // 密钥不符 (与 PC 端一致)
            Quote q;
            string text = UnpackPayload(pt, out q);              // 拆 {"t":..,"q":..} 引用包装
            NotifyMessage(rnick ?? "?", text, rts, q);
        } else if (type == "online") {
            if (rid != docId && rnick != null) { users[rid] = rnick; NotifyUsers(); }
        } else if (type == "typing") {
            if (rid != docId && rnick != null) {
                typingUsers.Add(rnick);
                NotifyTyping(rnick + " 正在输入…");
                System.Threading.Timer t = null;
                t = new System.Threading.Timer(delegate(object s) { typingUsers.Remove(rnick); try { t.Dispose(); } catch {} }, null, 4000, Timeout.Infinite);
            }
        } else if (type == "file-start") {
            if (JsonGet(json, "sid") != docId) OnFileStart(json);
        } else if (type == "file-chunk") {
            if (JsonGet(json, "sid") != docId) OnFileChunk(json);
        } else if (type == "file-end") {
            if (JsonGet(json, "sid") != docId) OnFileEnd(json);
        } else if (type == "file-resend") {
            OnFileResend(json);
        }
    }

    // ---------- Active Rooms 注册表 (仅 MQTT 模式) ----------
    class RoomEntry { public Dictionary<string, bool> nicks = new Dictionary<string, bool>(); public long lastTs; }
    static Dictionary<string, RoomEntry> activeRooms = new Dictionary<string, RoomEntry>();
    static List<string> roomNames = new List<string>();               // 与 roomList.Items 平行

    static void OnRegistryMessage(string json)
    {
        if (JsonGet(json, "type") != "room-beacon") return;
        string rid = JsonGet(json, "id");
        if (rid == null || rid == docId) return;
        string r = JsonGet(json, "room");
        if (r == null || r.Length == 0) return;
        string n = JsonGet(json, "nick");
        RoomEntry e;
        if (!activeRooms.TryGetValue(r, out e)) { e = new RoomEntry(); activeRooms[r] = e; }
        if (n != null) e.nicks[n] = true;
        string s = JsonGet(json, "ts"); long ts;
        e.lastTs = (s != null && long.TryParse(s, out ts)) ? ts : DateTimeOffset.Now.ToUnixTimeMilliseconds();
        NotifyRooms();
    }
    static bool ExpireRooms()
    {
        long now = DateTimeOffset.Now.ToUnixTimeMilliseconds();
        var dead = new List<string>();
        foreach (var kv in activeRooms) if (now - kv.Value.lastTs > 25000) dead.Add(kv.Key);
        foreach (string k in dead) activeRooms.Remove(k);
        return dead.Count > 0;
    }

    static void SendFileBackground(string path) { var t = new Thread(delegate() { SendFileWorker(path); }); t.IsBackground = true; t.Start(); }
    static void SendFileWorker(string path)
    {
        try {
            var fi = new FileInfo(path);
            if (fi.Length <= 0 || fi.Length > FILE_MAX) { NotifyEvent("文件为空或超过 2MB (无直连通道): " + fi.Name); return; }
            string b64 = Convert.ToBase64String(File.ReadAllBytes(path));
            string name = fi.Name;
            string fid = "f" + To36(DateTimeOffset.Now.ToUnixTimeMilliseconds()) + new Random().Next(99);
            int chunkSize = 8000;
            int total = (b64.Length + chunkSize - 1) / chunkSize;
            sentFiles[fid] = new SentFile { id = fid, b64 = b64, chunkSize = chunkSize, total = total };
            SendJson("{\"type\":\"file-start\",\"nick\":\"" + JsonEsc(nick) + "\",\"sid\":\"" + JsonEsc(docId) + "\",\"caption\":\"\",\"name\":\"" + JsonEsc(name) + "\",\"size\":" + fi.Length + ",\"id\":\"" + fid + "\",\"total\":" + total + "}");
            for (int i = 0; i < total; i++) {
                int len = Math.Min(chunkSize, b64.Length - i * chunkSize);
                string cd = b64.Substring(i * chunkSize, len);
                SendJson("{\"type\":\"file-chunk\",\"id\":\"" + fid + "\",\"idx\":" + i + ",\"seq\":" + i + ",\"total\":" + total + ",\"sid\":\"" + JsonEsc(docId) + "\",\"data\":\"" + cd + "\",\"len\":" + len + "}");
                Thread.Sleep(3);
            }
            SendJson("{\"type\":\"file-end\",\"id\":\"" + fid + "\",\"total\":" + total + ",\"sid\":\"" + JsonEsc(docId) + "\"}");
            NotifyEvent("已发送文件: " + name + " (" + (fi.Length / 1024) + "KB)");
        } catch (Exception ex) { NotifyEvent("发送文件失败: " + ex.Message); }
    }
    static void OnFileStart(string json)
    {
        string id = JsonGet(json, "id");
        if (id == null || pendingFiles.ContainsKey(id)) return;
        string ts = JsonGet(json, "total"), sz = JsonGet(json, "size");
        int total; long size;
        if (ts == null || !int.TryParse(ts, out total)) return;
        if (sz == null || !long.TryParse(sz, out size)) return;
        if (total > 20000 || size > 200L * 1024 * 1024) { NotifyEvent("拒绝超大文件: " + JsonGet(json, "name")); return; }
        var fr = new FileRecv { id = id, name = JsonGet(json, "name") ?? "file", sid = JsonGet(json, "sid"), sender = JsonGet(json, "nick") ?? "?", caption = JsonGet(json, "caption"), size = size, total = total, chunks = new string[total], t0 = DateTime.Now };
        pendingFiles[id] = fr;
        NotifyFileProgress(fr);
    }
    static void OnFileChunk(string json)
    {
        string id = JsonGet(json, "id");
        if (id == null) return;
        FileRecv fr;
        if (!pendingFiles.TryGetValue(id, out fr) || fr.done) return;
        string sid = JsonGet(json, "sid");
        if (sid != null && fr.sid != null && sid.Length > 0 && fr.sid.Length > 0 && sid != fr.sid) return;
        string data = JsonGet(json, "data");
        if (data == null) return;
        string lens = JsonGet(json, "len"); int len;
        if (lens != null && int.TryParse(lens, out len) && len != data.Length) return;
        string idxs = JsonGet(json, "idx"), seqs = JsonGet(json, "seq");
        int idx; if (idxs == null || !int.TryParse(idxs, out idx) || idx < 0 || idx >= fr.total) return;
        int seq; if (seqs != null && int.TryParse(seqs, out seq) && seq != idx) return;   // 重发的块不带 seq
        if (fr.chunks[idx] == null) fr.received++;
        fr.chunks[idx] = data;
        if (fr.received >= fr.total) CompleteFile(fr);
    }
    static void OnFileEnd(string json)
    {
        string id = JsonGet(json, "id");
        if (id == null) return;
        FileRecv fr;
        if (!pendingFiles.TryGetValue(id, out fr) || fr.done) return;
        if (fr.retries >= 5) { fr.failed = true; pendingFiles.Remove(id); NotifyFileProgress(fr); return; }
        fr.retries++;
        RequestResend(fr);
    }
    static void RequestResend(FileRecv fr)
    {
        var sb = new StringBuilder();
        for (int i = 0; i < fr.total; i++) if (fr.chunks[i] == null) { if (sb.Length > 0) sb.Append(','); sb.Append(i); }
        if (sb.Length == 0) return;
        SendJson("{\"type\":\"file-resend\",\"id\":\"" + fr.id + "\",\"missing\":[" + sb + "],\"room\":\"" + JsonEsc(room) + "\"}");
    }
    static void CompleteFile(FileRecv fr)
    {
        int cnt = 0;
        for (int i = 0; i < fr.total; i++) if (fr.chunks[i] != null) cnt++;
        if (cnt < fr.total) { fr.received = cnt; RequestResend(fr); return; }            // 有洞: 请求重发
        var sb = new StringBuilder();
        foreach (string c in fr.chunks) sb.Append(c);
        string b64 = sb.ToString();
        long expectLen = (long)Math.Ceiling(fr.size / 3.0) * 4;
        if (expectLen > 0 && Math.Abs(b64.Length - expectLen) > expectLen / 200 + 2) {   // 重组校验失败: 全量重发
            Array.Clear(fr.chunks, 0, fr.chunks.Length); fr.received = 0; RequestResend(fr); return;
        }
        try { fr.data = Convert.FromBase64String(b64); fr.done = true; } catch { fr.failed = true; }
        NotifyFileProgress(fr);
    }
    static void OnFileResend(string json)
    {
        string id = JsonGet(json, "id");
        if (id == null) return;
        SentFile sf;
        if (!sentFiles.TryGetValue(id, out sf)) return;
        int mi = json.IndexOf("\"missing\"");
        if (mi < 0) return;
        int lb = json.IndexOf('[', mi); int rb = json.IndexOf(']', lb < 0 ? mi : lb);
        if (lb < 0 || rb < 0) return;
        foreach (string tok in json.Substring(lb + 1, rb - lb - 1).Split(',')) {
            int idx; if (!int.TryParse(tok.Trim(), out idx)) continue;
            int off = idx * sf.chunkSize;
            if (off >= sf.b64.Length) continue;
            int len = Math.Min(sf.chunkSize, sf.b64.Length - off);
            string cd = sf.b64.Substring(off, len);
            SendJson("{\"type\":\"file-chunk\",\"id\":\"" + sf.id + "\",\"idx\":" + idx + ",\"total\":" + sf.total + ",\"sid\":\"" + JsonEsc(docId) + "\",\"data\":\"" + cd + "\",\"len\":" + len + "}");
        }
        SendJson("{\"type\":\"file-end\",\"id\":\"" + sf.id + "\",\"total\":" + sf.total + ",\"sid\":\"" + JsonEsc(docId) + "\"}");
    }

    // ---------- minimal JSON helpers ----------
    static string JsonEsc(string s) { return s == null ? "" : s.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", "\\r").Replace("\n", "\\n"); }
    static string JsonGet(string json, string key)
    {
        string pat = "\"" + key + "\":";
        int i = json.IndexOf(pat);
        if (i < 0) return null;
        i += pat.Length;
        while (i < json.Length && char.IsWhiteSpace(json[i])) i++;
        if (i >= json.Length) return null;
        if (json[i] == '"') {
            var sb = new StringBuilder();
            i++;
            while (i < json.Length) {
                char c = json[i];
                if (c == '\\' && i + 1 < json.Length) {
                    char n = json[i + 1];
                    if (n == '"' || n == '\\' || n == '/') sb.Append(n);
                    else if (n == 'n') sb.Append('\n');
                    else if (n == 'r') sb.Append('\r');
                    else if (n == 't') sb.Append('\t');
                    else sb.Append(n);
                    i += 2;
                } else if (c == '"') { break; } else { sb.Append(c); i++; }
            }
            return sb.ToString();
        }
        int start = i;
        while (i < json.Length && (char.IsDigit(json[i]) || json[i] == '-' || json[i] == '.' || json[i] == 'e' || json[i] == 'E' || json[i] == '+')) i++;
        return json.Substring(start, i - start);
    }
    // 提取 "key":{...} 的原始对象子串 (字符串感知的大括号配对)
    static string JsonGetObj(string json, string key)
    {
        string pat = "\"" + key + "\":";
        int i = json.IndexOf(pat);
        if (i < 0) return null;
        i += pat.Length;
        while (i < json.Length && char.IsWhiteSpace(json[i])) i++;
        if (i >= json.Length || json[i] != '{') return null;
        int depth = 0, start = i;
        bool inStr = false;
        for (; i < json.Length; i++) {
            char c = json[i];
            if (inStr) {
                if (c == '\\') i++;
                else if (c == '"') inStr = false;
            } else if (c == '"') inStr = true;
            else if (c == '{') depth++;
            else if (c == '}') { depth--; if (depth == 0) return json.Substring(start, i - start + 1); }
        }
        return null;
    }

    // ---------- quote (§3.5: 有引用时加密前明文 = {"t":正文,"q":{nick,text,q?}} ) ----------
    static string QuoteJson(Quote q, int depth)
    {
        if (q == null || depth >= 4) return "null";
        string s = "{\"nick\":\"" + JsonEsc(q.nick) + "\",\"text\":\"" + JsonEsc(q.text) + "\"";
        if (q.q != null) s += ",\"q\":" + QuoteJson(q.q, depth + 1);
        return s + "}";
    }
    static string PackPayload(string text, Quote q)
    {
        if (q == null) return text;
        return "{\"t\":\"" + JsonEsc(text) + "\",\"q\":" + QuoteJson(q, 0) + "}";
    }
    static Quote ParseQuote(string raw, int depth)
    {
        if (raw == null || depth >= 4) return null;
        var q = new Quote();
        q.nick = JsonGet(raw, "nick");
        q.text = JsonGet(raw, "text");
        if (q.nick == null && q.text == null) return null;
        q.q = ParseQuote(JsonGetObj(raw, "q"), depth + 1);
        return q;
    }
    // 解包: 明文若是 {"t":..(字符串),"q":..} 则拆出正文与引用, 否则原样返回
    static string UnpackPayload(string pt, out Quote quote)
    {
        quote = null;
        if (pt == null) return null;
        if (pt.StartsWith("{")) {
            string t = JsonGet(pt, "t");
            if (t != null && JsonGetObj(pt, "t") == null) {
                quote = ParseQuote(JsonGetObj(pt, "q"), 0);
                return t;
            }
        }
        return pt;
    }
    static string QuoteText(Quote q)
    {
        var sb = new StringBuilder();
        int d = 0;
        while (q != null && d < 4) {
            if (d > 0) sb.Append(" ▸ ");
            string t = q.text ?? "";
            if (t.Length > 40) t = t.Substring(0, 40) + "…";
            sb.Append(q.nick).Append(": ").Append(t);
            q = q.q; d++;
        }
        return sb.ToString();
    }

    static void ShowHistory()
    {
        List<string> l;
        if (!history.TryGetValue(room, out l) || l.Count == 0) return;
        AddMsgLine("· —— 历史消息 ——", null);
        int from = l.Count > 80 ? l.Count - 80 : 0;
        for (int i = from; i < l.Count; i++) {
            string line = l[i];
            int p1 = line.IndexOf('|'); if (p1 < 0) continue;
            int p2 = line.IndexOf('|', p1 + 1); if (p2 < 0) continue;
            string ts = line.Substring(0, p1);
            string n = line.Substring(p1 + 1, p2 - p1 - 1);
            string txt;
            try { txt = Encoding.UTF8.GetString(Convert.FromBase64String(line.Substring(p2 + 1))); } catch { continue; }
            long msl; string tm = long.TryParse(ts, out msl) ? DateTimeOffset.FromUnixTimeMilliseconds(msl).LocalDateTime.ToString("MM-dd HH:mm") : "";
            AddMsgLine(n + "  " + tm + "  " + txt, new MsgMeta { nick = n, text = txt });
        }
    }

    // ---------- message history (每房间最近 80 条) ----------
    static string HistPath() { return Path.Combine(CfgDir(), "chat-history.txt"); }
    static void HistAdd(string r, string n, string ts, string t)
    {
        try {
            List<string> l;
            if (!history.TryGetValue(r, out l)) { l = new List<string>(); history[r] = l; }
            l.Add((ts ?? "") + "|" + (n ?? "").Replace("|", "_") + "|" + Convert.ToBase64String(Encoding.UTF8.GetBytes(t ?? "")));
            if (l.Count > 80) l.RemoveRange(0, l.Count - 80);
            if (++histDirty >= 10) { histDirty = 0; SaveHistory(); }
        } catch {}
    }
    static void LoadHistory()
    {
        try {
            history.Clear();
            if (!File.Exists(HistPath())) return;
            foreach (string raw in File.ReadAllLines(HistPath(), Encoding.UTF8)) {
                int t1 = raw.IndexOf('\t'); if (t1 < 1) continue;
                string r = raw.Substring(0, t1);
                List<string> l;
                if (!history.TryGetValue(r, out l)) { l = new List<string>(); history[r] = l; }
                l.Add(raw.Substring(t1 + 1));
            }
        } catch {}
    }
    static void SaveHistory()
    {
        try {
            Directory.CreateDirectory(CfgDir());
            var sb = new StringBuilder();
            int rooms = 0;
            foreach (var kv in history) {
                if (rooms++ >= 30) break;
                var l = kv.Value;
                int from = l.Count > 80 ? l.Count - 80 : 0;
                string r = kv.Key.Replace('\t', ' ');
                for (int i = from; i < l.Count; i++) sb.Append(r).Append('\t').Append(l[i]).Append("\r\n");
            }
            File.WriteAllText(HistPath(), sb.ToString(), new UTF8Encoding(false));
        } catch {}
    }

    // ---------- persisted ----------
    static void LoadCfg()
    {
        try {
            if (!File.Exists(CfgPath())) return;
            foreach (string raw in File.ReadAllLines(CfgPath(), Encoding.UTF8)) {
                string t = raw.Trim(); int eq = t.IndexOf('='); if (eq < 1) continue;
                string k = t.Substring(0, eq).Trim().ToLower(), v = t.Substring(eq + 1).Trim();
                if (k == "nick") nick = v;
                else if (k == "room") room = v;
                else if (k == "key") customKey = v;
                else if (k == "broker") brokerSel = v;
                else if (k == "lastbroker") lastBroker = v;
            }
        } catch {}
    }
    static void SaveCfg()
    {
        try {
            Directory.CreateDirectory(CfgDir());
            File.WriteAllText(CfgPath(), "nick = " + nick + "\r\nroom = " + room + "\r\nkey = " + customKey + "\r\nbroker = " + brokerSel + "\r\nlastbroker = " + lastBroker + "\r\n", new UTF8Encoding(false));
        } catch {}
    }
