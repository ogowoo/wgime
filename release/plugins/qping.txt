code = qping
name = QuickPing
desc = Interactive console ping test with manual parameters (PS)

[psx]
Write-Host "=== QuickPing Parameter Setup ===" -ForegroundColor Cyan

$Target = Read-Host "Target (e.g. 192.168.1.1 or baidu.com)"
if ([string]::IsNullOrWhiteSpace($Target)) {
    Write-Host "Target cannot be empty." -ForegroundColor Red
    exit 1
}

$CountInput = Read-Host "Count [100]"
$Count = if ($CountInput) { [int]$CountInput } else { 100 }

$IntervalInput = Read-Host "Interval ms [100]"
$Interval = if ($IntervalInput) { [int]$IntervalInput } else { 100 }

$TimeoutInput = Read-Host "Timeout ms [1000]"
$Timeout = if ($TimeoutInput) { [int]$TimeoutInput } else { 1000 }

$QuietInput = Read-Host "Quiet mode (y/N)"
$Quiet = $QuietInput -match '^[Yy]'

Write-Host ""
Write-Host "Pinging $Target ($Count packets, ${Interval}ms interval, ${Timeout}ms timeout)..." -ForegroundColor Yellow
Write-Host ""

$ping = New-Object System.Net.NetworkInformation.Ping
$rtts = New-Object System.Collections.Generic.List[double]
$sent = 0
$received = 0
$lost = 0

try {
    for ($i = 1; $i -le $Count; $i++) {
        $sent++

        try {
            $reply = $ping.Send($Target, $Timeout)

            if ($reply.Status -eq [System.Net.NetworkInformation.IPStatus]::Success) {
                $rtt = [double]$reply.RoundtripTime
                $rtts.Add($rtt)
                $received++

                if (-not $Quiet) {
                    Write-Host ("{0,4}  RTT={1,5} ms" -f $i, $rtt) -ForegroundColor Green
                }
            }
            else {
                $lost++
                if (-not $Quiet) {
                    Write-Host ("{0,4}  {1}" -f $i, $reply.Status) -ForegroundColor Red
                }
            }
        }
        catch {
            $lost++
            if (-not $Quiet) {
                Write-Host ("{0,4}  ERROR: {1}" -f $i, $_.Exception.Message) -ForegroundColor Red
            }
        }

        if (($i -lt $Count) -and ($Interval -gt 0)) {
            Start-Sleep -Milliseconds $Interval
        }
    }
}
finally {
    $ping.Dispose()
}

$lossPct = [math]::Round(($lost * 100.0 / $sent), 2)

Write-Host ""
Write-Host "========== RESULT ==========" -ForegroundColor Cyan

if ($rtts.Count -eq 0) {
    [pscustomobject][ordered]@{
        Target        = $Target
        Sent          = $sent
        Received      = $received
        Lost          = $lost
        LossPct       = $lossPct
        MinRTT_ms     = $null
        AvgRTT_ms     = $null
        MaxRTT_ms     = $null
        P95_ms        = $null
        P99_ms        = $null
        StdDev_ms     = $null
        AvgJitter_ms  = $null
        MaxJitter_ms  = $null
    } | Format-List
}
else {
    $measure = $rtts | Measure-Object -Minimum -Maximum -Average
    $minRtt = [double]$measure.Minimum
    $avgRtt = [double]$measure.Average
    $maxRtt = [double]$measure.Maximum

    $jitters = New-Object System.Collections.Generic.List[double]
    for ($i = 1; $i -lt $rtts.Count; $i++) {
        $delta = [math]::Abs($rtts[$i] - $rtts[$i - 1])
        $jitters.Add($delta)
    }

    if ($jitters.Count -gt 0) {
        $jitterMeasure = $jitters | Measure-Object -Average -Maximum
        $avgJitter = [double]$jitterMeasure.Average
        $maxJitter = [double]$jitterMeasure.Maximum
    }
    else {
        $avgJitter = 0.0
        $maxJitter = 0.0
    }

    $sumSquaredDeviation = 0.0
    foreach ($rttValue in $rtts) {
        $difference = $rttValue - $avgRtt
        $sumSquaredDeviation += $difference * $difference
    }
    $stdDev = [math]::Sqrt($sumSquaredDeviation / $rtts.Count)

    [double[]]$sorted = $rtts | Sort-Object
    $p95Index = [math]::Ceiling(0.95 * $sorted.Count) - 1
    $p99Index = [math]::Ceiling(0.99 * $sorted.Count) - 1
    $p95 = $sorted[$p95Index]
    $p99 = $sorted[$p99Index]

    [pscustomobject][ordered]@{
        Target        = $Target
        Sent          = $sent
        Received      = $received
        Lost          = $lost
        LossPct       = $lossPct
        MinRTT_ms     = [math]::Round($minRtt, 2)
        AvgRTT_ms     = [math]::Round($avgRtt, 2)
        MaxRTT_ms     = [math]::Round($maxRtt, 2)
        P95_ms        = [math]::Round($p95, 2)
        P99_ms        = [math]::Round($p99, 2)
        StdDev_ms     = [math]::Round($stdDev, 2)
        AvgJitter_ms  = [math]::Round($avgJitter, 2)
        MaxJitter_ms  = [math]::Round($maxJitter, 2)
    } | Format-List
}
[/psx]
