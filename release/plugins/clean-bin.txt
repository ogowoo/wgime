code = qls
name = 清空回收站
desc = confirm 后调用 PowerShell Clear-RecycleBin
version = 1.0
author = ogowoo
perm = destructive

[powershell]
Clear-RecycleBin -Force -ErrorAction SilentlyContinue
Write-Output "recycle bin cleared"
[/powershell]
